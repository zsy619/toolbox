Chapter 02

数据模型

ts
// 编辑器核心数据
type Project = {
  id: string;
  tracks: Track[];
  duration: number;  // 帧
  composition: { width: number; height: number; fps: number };
};
type Track = {
  id: string;
  type: 'video' | 'audio' | 'text' | 'shape';
  clips: Clip[];
};
type Clip = {
  id: string;
  trackId: string;
  from: number;      // 起始帧
  durationInFrames: number;
  src?: string;      // 视频/图片 URL
  text?: string;     // 文本
  props?: Record<string, any>;
};
Chapter 03

PlayerRef 控制时间轴

tsx
import { Player, PlayerRef } from '@remotion/player';

const ref = useRef<PlayerRef>(null);

// 点击时间轴某帧
const seek = (frame: number) => ref.current?.seekTo(frame);

// 实时同步到 Zustand store
useEffect(() => {
  ref.current?.addEventListener('frameupdate', e => {
    setStore({ currentFrame: e.detail.frame });
  });
}, []);
Chapter 04

轨道渲染

把 Project 数据映射到 Composition 子树:

tsx
const Editor: React.FC<{project: Project}> = ({ project }) => (
  <Player
    component={VideoRenderer}
    durationInFrames={project.duration}
    compositionWidth={project.composition.width}
    compositionHeight={project.composition.height}
    fps={project.composition.fps}
    ref={ref}
  />
);

const VideoRenderer: React.FC = () => project.tracks.map(track => (
  <AbsoluteFill>
    {track.clips.map(clip => (
      <Sequence from={clip.from} durationInFrames={clip.durationInFrames}>
        {clip.type === 'video' && <OffthreadVideo src={clip.src} />}
        {clip.type === 'text' && <TextOverlay text={clip.text} />}
      </Sequence>
    ))}
  </AbsoluteFill>
));
Chapter 05

导出到 Lambda

ts
// 点击"导出"按钮
const exportVideo = async () => {
  setStatus('rendering');
  const { renderId, url } = await renderMediaOnLambda({
    composition: 'Editor',
    serveUrl: SERVE_URL,
    functionName: 'editor-render',
    region: 'us-east-1',
    inputProps: { project, projectId },
    webhook: { url: `${API}/webhook/${projectId}`,
               secret: process.env.WEBHOOK_SECRET! },
  });
  setRenderId(renderId);
};

// 客户端轮询 OR WebSocket 接收完成
useEffect(() => {
  const ws = new WebSocket(`${WS}/render/${projectId}`);
  ws.onmessage = e => {
    const { status, url } = JSON.parse(e.data);
    if (status === 'done') setStatus('done');
  };
  return () => ws.close();
}, [projectId]);

const doc = new Y.Doc();

);

// project 用 Y.Map,clip 用 Y.Array

yTracks.observeDeep(() => setProject(yTracks.toJSON()));
